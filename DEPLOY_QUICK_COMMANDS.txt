# 1) Upload từ Windows PowerShell
scp .\hachiweb3001-tra-ma-v1-deploy.tar.gz root@IP_VPS:/tmp/

# 2) SSH
ssh root@IP_VPS

# 3) Giải nén vào thư mục project
cd /var/www/hachiweb3001
mkdir -p backups
cp -a toa-hang/backend/data/toa-hang.db "backups/toa-hang-$(date +%Y%m%d-%H%M%S).db"
cp -a toa-hang/backend/.env "backups/backend-env-$(date +%Y%m%d-%H%M%S)" 2>/dev/null || true
tar -xzf /tmp/hachiweb3001-tra-ma-v1-deploy.tar.gz -C /var/www/hachiweb3001

# 4A) PM2 — sửa tên process nếu khác toa-hang
APP_DIR=/var/www/hachiweb3001/toa-hang PM2_NAME=toa-hang bash deploy_tra_ma_v1.sh

# 4B) Hoặc systemd
# APP_DIR=/var/www/hachiweb3001/toa-hang SYSTEMD_SERVICE=toa-hang.service bash deploy_tra_ma_v1.sh

# 5) VPS offline, chỉ khi node_modules backend hiện tại đã đủ
# APP_DIR=/var/www/hachiweb3001/toa-hang PM2_NAME=toa-hang SKIP_INSTALL=1 SKIP_BUILD=1 bash deploy_tra_ma_v1.sh

# 6) Kiểm tra
curl -s http://127.0.0.1:3001/api/health
pm2 logs toa-hang --lines 100
